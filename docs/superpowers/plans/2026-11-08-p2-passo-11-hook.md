```python
"""Guarda de CPF/CNPJ: acha números de contribuinte num texto sem nunca repeti-los.

É o ponto único usado pelo hook do Claude Code (passo 11) e pelo server
(passo 10): `find_in_arguments` varre os argumentos de uma chamada para os
dois. Um "achado" exige formato E dígitos verificadores corretos: só a regex
confundiria números de processo, datas e artigos com documentos.
"""

from __future__ import annotations

import re
from collections import Counter
from dataclasses import dataclass
from typing import Literal

Kind = Literal["cpf", "cnpj"]

# Pesos do módulo 11 do CNPJ ("Perguntas & Respostas: CNPJ alfanumérico" da
# RFB, pergunta 14). Valem igual para o CNPJ numérico e o alfanumérico.
CNPJ_DV1_WEIGHTS = (5, 4, 3, 2, 9, 8, 7, 6, 5, 4, 3, 2)
CNPJ_DV2_WEIGHTS = (6, 5, 4, 3, 2, 9, 8, 7, 6, 5, 4, 3, 2)


def _mod11_digit(values: list[int], weights: tuple[int, ...]) -> int:
    """Um dígito verificador: soma ponderada, resto por 11; resto 0 ou 1 vira 0."""
    remainder = sum(v * w for v, w in zip(values, weights, strict=True)) % 11
    return 0 if remainder < 2 else 11 - remainder


def cpf_check_digits(base9: str) -> str:
    """Os 2 DVs do CPF para uma base de 9 dígitos (pesos 10..2 e depois 11..2)."""
    if not re.fullmatch(r"[0-9]{9}", base9):
        raise ValueError("a base do CPF deve ter exatamente 9 dígitos de 0 a 9")
    values = [int(c) for c in base9]
    dv1 = _mod11_digit(values, tuple(range(10, 1, -1)))
    dv2 = _mod11_digit(values + [dv1], tuple(range(11, 1, -1)))
    return f"{dv1}{dv2}"


def cnpj_check_digits(base12: str) -> str:
    """Os 2 DVs do CNPJ para uma base de 12 caracteres (0-9 e A-Z maiúsculas).

    Cada caractere vale o código ASCII menos 48: os dígitos mantêm o valor de
    sempre ('7' -> 7) e as letras passam a valer 17 ('A') a 42 ('Z').
    """
    if not re.fullmatch(r"[0-9A-Z]{12}", base12):
        raise ValueError("a base do CNPJ deve ter 12 caracteres entre 0-9 e A-Z")
    values = [ord(c) - 48 for c in base12]
    dv1 = _mod11_digit(values, CNPJ_DV1_WEIGHTS)
    dv2 = _mod11_digit(values + [dv1], CNPJ_DV2_WEIGHTS)
    return f"{dv1}{dv2}"


@dataclass(frozen=True)
class Finding:
    """Um achado: só o tipo e a posição no texto. Nunca guarda o número."""

    kind: Kind
    start: int
    end: int


# Fronteiras: o candidato não pode estar colado a letra/dígito, nem a "." ou
# "-" com um DÍGITO do outro lado (o número continuaria). Assim
# "1.234.567.890-12" não vira o CPF "234.567.890-12", mas "CPF 000.000.000-00."
# (ponto final) ainda é candidato.
# Passo 11: a "/" não conta como continuação (numa URL ou data ela separa
# campos; a barra do próprio CNPJ fica dentro do match), e letra do outro lado
# de "." ou "-" também não. Antes, "https://exemplo.invalid/<cnpj>" e
# "cpf-<número>" passavam, e o WebFetch recebe exatamente URLs.
# [0-9] e não \d: em Python, \d também casa dígitos de outros alfabetos.
_BEFORE = r"(?<![0-9A-Za-z])(?<![0-9][.-])"
_AFTER = r"(?![0-9A-Za-z])(?![.-][0-9])"
# Cada separador é opcional sozinho: a RFB mostra máscaras parciais como
# "AA345678/0003-29" (Perguntas & Respostas, pergunta 23).
_CPF_RE = re.compile(_BEFORE + r"[0-9]{3}\.?[0-9]{3}\.?[0-9]{3}-?[0-9]{2}" + _AFTER)
# IGNORECASE: a RFB só emite maiúsculas, mas a guarda prefere bloquear um
# "ab.cde.fgh/0001-xx" digitado em minúsculas a deixá-lo passar.
_CNPJ_RE = re.compile(
    _BEFORE + r"[0-9A-Z]{2}\.?[0-9A-Z]{3}\.?[0-9A-Z]{3}/?[0-9A-Z]{4}-?[0-9]{2}" + _AFTER,
    re.IGNORECASE,
)
_SEPARATORS = re.compile(r"[./-]")


def _is_valid(compact: str, kind: Kind) -> bool:
    """DVs corretos e não é um caractere repetido (000..., 111...: DV "válido" por acaso)."""
    if len(set(compact)) == 1:
        return False
    if kind == "cpf":
        return compact[9:] == cpf_check_digits(compact[:9])
    return compact[12:] == cnpj_check_digits(compact[:12])


def find_taxpayer_ids(text: str) -> list[Finding]:
    """Todos os CPF/CNPJ com DV válido em `text`, em ordem de posição."""
    findings: list[Finding] = []
    for kind, pattern in (("cpf", _CPF_RE), ("cnpj", _CNPJ_RE)):
        for match in pattern.finditer(text):
            compact = _SEPARATORS.sub("", match.group()).upper()
            if _is_valid(compact, kind):
                findings.append(Finding(kind=kind, start=match.start(), end=match.end()))
    return sorted(findings, key=lambda f: f.start)


def describe_findings(findings: list[Finding]) -> str:
    """Mensagem para humano ou modelo: quantos e de que tipo, sem nenhum número."""
    if not findings:
        return "nenhum CPF ou CNPJ encontrado"
    counts = Counter(f.kind for f in findings)
    parts = [f"{counts[k]} {k.upper()}" for k in ("cpf", "cnpj") if counts[k]]
    return "possível dado de contribuinte: " + " e ".join(parts) + " (números omitidos)"


def _scan(value: object, path: str) -> str | None:
    """Motivo para o primeiro texto com CPF/CNPJ válido sob `value` (chaves também), ou None."""
    if isinstance(value, str):
        findings = find_taxpayer_ids(value)
        return f"{describe_findings(findings)} no campo '{path}'" if findings else None
    if isinstance(value, dict):
        for key, item in value.items():
            found = _scan(str(key), f"{path}.<chave>") or _scan(item, f"{path}.{key}")
            if found:
                return found
    if isinstance(value, list | tuple):
        for i, item in enumerate(value):
            found = _scan(item, f"{path}[{i}]")
            if found:
                return found
    return None


def find_in_arguments(arguments: dict) -> str | None:
    """Varre os argumentos de uma chamada de ferramenta: todos os textos, em qualquer profundidade.

    Usada pelo server (antes da validação do SDK) e pelo hook do Claude Code:
    as duas guardas não divergem. Chaves também são varridas, porque um número
    pode chegar como nome de campo. Devolve o motivo (tipo, quantidade e
    caminho do campo, nunca o número) ou None. Caminhos: `consulta`,
    `extra.lista[1]`, `<chave>` (nome de campo do topo), `extra.<chave>`.
    """
    for key, value in arguments.items():
        found = _scan(str(key), "<chave>") or _scan(value, str(key))
        if found:
            return found
    return None
```

```python
        super().__init__(json.dumps({"erro": code, "mensagem": message}, ensure_ascii=False))


class TaxLawServer(MCPServer):
    """MCPServer com a guarda de CPF/CNPJ antes da validação do SDK."""

    async def call_tool(self, name, arguments, context=None):  # type: ignore[override]
        # A mesma varredura do hook do Claude Code (guard.find_in_arguments).
        found = find_in_arguments(arguments or {})
        if found:
            raise TaxLawError("dados_de_contribuinte", found)
        try:
            return await super().call_tool(name, arguments, context)
        except SDKToolError as exc:
            # O SDK embrulha o erro da ferramenta em "Error executing tool <nome>: ...";
            # desembrulhamos para o texto ser só o JSON, igual ao da guarda acima.
            if isinstance(exc.__cause__, TaxLawError):
                raise exc.__cause__ from None
            if isinstance(exc.__cause__, ValidationError):
                # Só o nome dos campos: o valor rejeitado é dado do chamador.
                fields = sorted({".".join(str(p) for p in err["loc"]) for err in exc.__cause__.errors()})
                raise TaxLawError("entrada_invalida", "argumento inválido: " + ", ".join(fields)) from None
            raise


def _call(function, **kwargs) -> Any:
    """Chama a função pura e converte `tools.ToolError` no erro do SDK (isError)."""
    try:
        return function(**kwargs)
    except tools.ToolError as exc:
        raise TaxLawError(exc.code, exc.message) from None


DATE_FIELD = Field(
    default=None,
    description="Data AAAA-MM-DD em que a pergunta se coloca. Ausente: a data de hoje (ecoada na resposta).",
)


def build_server(ctx: tools.Context | None = None, *, today: date | None = None) -> TaxLawServer:
    """Monta o server. `ctx` e `today` existem para os testes; em produção, o corpus real e hoje."""
    context = ctx or tools.load_context()
    server = TaxLawServer(SERVER_NAME, version="0.1.0")

    @server.tool(annotations=READ_ONLY)
    def buscar_dispositivos(
        consulta: Annotated[str, Field(description="Palavras a buscar no texto da EC 132/2023 e dos dispositivos da CF e do ADCT que ela inclui ou altera (1 a 500 caracteres).")],
        data_referencia: Annotated[str | None, DATE_FIELD] = None,
        # strict: sem isso o pydantic aceitaria True e "5" como inteiros, e o
        # server divergiria de tools.py, que os rejeita.
        limite: Annotated[int, Field(strict=True, description="Número máximo de resultados, de 1 a 10.")] = 5,
    ) -> BuscaSaida:
        """Busca dispositivos da reforma tributária (EC 132/2023) por palavras.

        Cada resultado traz a citação, um trecho, a versão do texto que vale na
        data de referência e as datas de efeito com fundamento. Sem resultado
        relevante: encontrado=false com motivo. Use obter_dispositivo para o
        texto completo antes de citar.
        """
        return _call(tools.buscar_dispositivos, ctx=context, consulta=consulta, data_referencia=data_referencia, limite=limite, today=today)

    @server.tool(annotations=READ_ONLY)
    def obter_dispositivo(
        id: Annotated[str, Field(description="Id do dispositivo, como devolvido pela busca (ex.: CF.art156-A.par1.incIX).")],
        data_referencia: Annotated[str | None, DATE_FIELD] = None,
        incluir_filhos: Annotated[bool, Field(strict=True, description="Incluir os filhos diretos (incisos, alíneas, parágrafos).")] = True,
    ) -> DispositivoSaida:
        """Texto completo de um dispositivo na versão que vale na data de referência.

        Traz citação, origem na EC, filhos diretos, datas de efeito com
        fundamento, se produz efeitos na data (true, false ou null quando não
        se sabe), as outras versões e observações (redação anterior fora do
        corpus, revogação). Id inexistente: encontrado=false.
        """
        return _call(tools.obter_dispositivo, ctx=context, id=id, data_referencia=data_referencia, incluir_filhos=incluir_filhos, today=today)

    return server


def main() -> None:
    build_server().run("stdio")


if __name__ == "__main__":
    main()
```

```python
"""Hook `PreToolUse` do Claude Code: bloqueia CPF/CNPJ antes de a chamada sair (critério 2).

O Claude Code roda o hook antes de cada chamada de ferramenta cujo nome casa
com o matcher em `.claude/settings.json` (as ferramentas `mcp__taxlaw__*`,
`WebSearch` e `WebFetch`). Ele recebe a chamada em JSON no stdin, com
`tool_name` e `tool_input` (os argumentos que o modelo gerou).

Contrato com o Claude Code (documentação de hooks, conferida em 2026-11-08):
- código 0: a chamada segue o fluxo normal de permissões;
- código 2: a chamada é bloqueada, e o stderr volta ao modelo como motivo;
- **qualquer outro código não bloqueia** (a chamada segue). Por isso este hook
  nunca termina com 1: entrada ilegível ou erro interno também dão 2. Na
  dúvida, bloqueia (fail closed).

O motivo diz o tipo, a quantidade e o campo, nunca o número.
"""

from __future__ import annotations

import json
import sys
from typing import TextIO

from taxlaw.guard import find_in_arguments

ALLOW = 0
BLOCK = 2

UNREADABLE = "não foi possível ler os argumentos da chamada; bloqueada por precaução"
INTERNAL_ERROR = "erro interno do hook; chamada bloqueada por precaução"


def check(payload: object) -> str | None:
    """Motivo para bloquear a chamada, ou None para deixá-la seguir."""
    if not isinstance(payload, dict) or not isinstance(payload.get("tool_input"), dict):
        return UNREADABLE
    return find_in_arguments(payload["tool_input"])


def main(stdin: TextIO | None = None, stderr: TextIO | None = None) -> int:
    """Lê a chamada do stdin e devolve o código de saída do hook."""
    stdin = stdin or sys.stdin
    stderr = stderr or sys.stderr
    try:
        try:
            payload = json.loads(stdin.read())
        except ValueError:  # JSON inválido (UnicodeDecodeError também é ValueError)
            payload = None
        reason = check(payload)
    except Exception:  # noqa: BLE001 - qualquer falha bloqueia; nunca sair com 1
        reason = INTERNAL_ERROR
    if reason is None:
        return ALLOW
    stderr.write(f"Chamada bloqueada pelo hook do projeto: {reason}. CPF e CNPJ não são enviados a ferramentas.\n")
    return BLOCK
```

````markdown
# Passo 11: o hook que bloqueia CPF/CNPJ (critério 2)

## O que mudou

- **`.claude/settings.json`** (novo, versionado) registra um hook `PreToolUse`.
  - O matcher `^(mcp__taxlaw__.*|WebSearch|WebFetch)$` usa âncoras porque o Claude Code testa o matcher como regex sem âncora. Sem elas, `WebSearch` casaria dentro de outros nomes.
  - O comando é `uv run --quiet --project "$CLAUDE_PROJECT_DIR" python "$CLAUDE_PROJECT_DIR/hooks/block_taxpayer_ids.py"`, com timeout de 30 s.
- **`hooks/block_taxpayer_ids.py`** é o ponto de entrada. A lógica está em `src/taxlaw/hook.py`, testável como função.
  - Ele lê a chamada do stdin e varre todos os textos de `tool_input`, inclusive chaves e valores aninhados.
  - Se achar CPF ou CNPJ, escreve o motivo no stderr (tipo e campo, sem o número) e sai com **2**.
- **Bloqueia na dúvida (fail closed).** Pela documentação, só o código 2 bloqueia: qualquer outro deixa a chamada passar. Por isso JSON ilegível, `tool_input` ausente, erro interno ou pacote não instalado também saem com 2.
- **`guard.find_in_arguments`.** A varredura de argumentos saiu do `server.py` e foi para a `guard.py`. O hook e o server usam a mesma função (spec, seção 4: "as duas guardas não divergem").
````
